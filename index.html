let gpsLocation = ""; // Collect GPS in background navigator.geolocation.getCurrentPosition( position => { gpsLocation = `${position.coords.latitude}, ${position.coords.longitude}`; }, () => { showToast("Unable to fetch location. Please try again.", 5000); // Longer duration for location error } ); function showToast(message, duration = 3000) { const toast = document.getElementById("toast"); toast.innerText = message; toast.className = "show"; setTimeout(() => { toast.className = toast.className.replace("show", ""); }, duration); // Duration is configurable (default 3000ms) } function track(action) { const phoneNumber = document.getElementById('phoneNumber').value; if (!phoneNumber) { showToast("Please enter your phone number."); return; } if (!gpsLocation) { showToast("Location not available yet. Try again shortly."); return; } // Call backend const data = { action: action, phone: phoneNumber, location: gpsLocation }; sendToBackend(data); } function submitLeaveRequest() { const phoneNumber = document.getElementById('phoneNumber').value; const leaveDate = document.getElementById('leaveDate').value; const leaveReason = document.getElementById('leaveReason').value; if (!phoneNumber || !leaveDate || !leaveReason) { showToast("Please fill in all fields."); return; } // Prepare data for leave request const data = { action: 'leave', phone: phoneNumber, leaveDate: leaveDate, reason: leaveReason }; sendToBackend(data); } function sendToBackend(data) { // Make a POST request to the Google Apps Script backend fetch('https://script.google.com/macros/s/YOUR_SCRIPT_ID/exec', { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded', }, body: new URLSearchParams(data) }) .then(response => response.text()) .then(responseText => { showToast(responseText); // Show backend response as toast notification }) .catch(error => { showToast('Error: ' + error.message); }); } function showLeaveForm() { document.getElementById('leaveForm').style.display = 'block'; document.getElementById('loginBtn').disabled = true; document.getElementById('logoutBtn').disabled = true; } function cancelLeaveRequest() { document.getElementById('leaveForm').style.display = 'none'; document.getElementById('loginBtn').disabled = false; document.getElementById('logoutBtn').disabled = false; document.getElementById('leaveDate').value = ''; document.getElementById('leaveReason').value = ''; }
